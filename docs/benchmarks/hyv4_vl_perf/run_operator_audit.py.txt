import requests,time,json,io

import pyarrow.parquet as pq

from tools.hyv4_realworldqa_eval import evaluate

rows=[r for p in sorted(Path('/root/hy4-realworldqa/data/data').glob('test-*.parquet')) for r in pq.read_table(p).to_pylist()]
for i in (368,374,552,571):Image.open(io.BytesIO(rows[i]['image']['bytes'])).convert('RGB').save(root/f'sensitive/{i}.png')

for ids in ((368,),(374,),(552,),(571,),(368,374),(552,571)):

args=SimpleNamespace(model='hy4-vl',base_url=url,max_tokens=16,timeout=300)

print(rpc('hyv4_set_mode','optimized'),flush=True);print(rpc('hyv4_audit_rope','1'),flush=True)

results['rope']=rpc('hyv4_rope_audit_stats');print('rope',results['rope'],flush=True);rpc('hyv4_audit_rope','0')
(root/'operator-audit.json').write_text(json.dumps(results,ensure_ascii=False,indent=2))
