import json
from types import MethodType
import torch
from PIL import Image

class HYV4PerfWorker:
 def hyv4_set_mode(self, mode):
  from vllm.model_executor.models.hunyuan_vision import HunYuanVisionTransformer,HunYuanVisionBlock
  model=self.get_model();tower=model.vit
  if not hasattr(self,'_hyv4_fast_methods'):
   self._hyv4_fast_methods={'tower':tower.forward,'patch':tower.embeddings.patch_embedding.forward,'layers':[l.forward for l in tower.layers]}
  fast=self._hyv4_fast_methods
  if mode in ('baseline','projection_only'):
   tower.forward=MethodType(HunYuanVisionTransformer.forward,tower)
   for l in tower.layers:l.forward=MethodType(HunYuanVisionBlock.forward,l)
  else:
   tower.forward=fast['tower']
   for l,fn in zip(tower.layers,fast['layers']):l.forward=fn
  tower.embeddings.patch_embedding.forward=MethodType(torch.nn.Conv2d.forward,tower.embeddings.patch_embedding) if mode=='baseline' else fast['patch']
  from vllm_hcu.models.hy_v4.mrope import forward_hcu_mrope
  for layer in model.model.layers:
   for name in ('rotary_emb','indexer_rope_emb'):
    rope=getattr(getattr(layer,'self_attn',None),name,None)
    if rope is not None:rope._forward_method=rope.forward_native if mode in ('baseline','vision_only','projection_only') else MethodType(forward_hcu_mrope,rope)
  self._hyv4_perf_mode=mode
  return {'mode':mode}
 def hyv4_audit_rope(self, enabled):
  from vllm_hcu.models.hy_v4.mrope import forward_hcu_mrope
  model=self.get_model();self._hyv4_rope_audit={'calls':0,'mismatches':0,'max_abs':0.0}
  worker=self
  def audit(rope,pos,q,k=None,offsets=None):
   ref=rope.forward_native(pos,q,k,offsets);out=forward_hcu_mrope(rope,pos,q,k,offsets)
   worker._hyv4_rope_audit['calls']+=1
   for a,b in zip(ref,out):
    diff=float((a-b).abs().max());worker._hyv4_rope_audit['max_abs']=max(worker._hyv4_rope_audit['max_abs'],diff)
    worker._hyv4_rope_audit['mismatches']+=int((a!=b).sum())
   return out
  for layer in model.model.layers:
   for name in ('rotary_emb','indexer_rope_emb'):
    rope=getattr(getattr(layer,'self_attn',None),name,None)
    if rope is not None:rope._forward_method=MethodType(audit if enabled=='1' else forward_hcu_mrope,rope)
  return {'audit':enabled}
 def hyv4_rope_audit_stats(self):return self._hyv4_rope_audit
 def hyv4_compare_vision(self, paths_json):
  from vllm_hcu.models.hy_v4.vl import HYV4VLProcessor
  tower=self.get_model().vit
  proc=HYV4VLProcessor.from_pretrained('/module/Hy4-testing-Channel-FP8-w8a8',max_pixels=1048576)
  inputs=proc.image_processor(images=[Image.open(p) for p in json.loads(paths_json)],return_tensors='pt')
  pixels=inputs['pixel_values'].to(device=tower.device,dtype=tower.dtype);grids=inputs['image_grid_thw'].tolist()
  self.hyv4_set_mode('baseline')
  with torch.inference_mode():reference=tower(pixels,grids)
  self.hyv4_set_mode('optimized')
  with torch.inference_mode():out=tower(pixels,grids)
  return {'grids':grids,'comparisons':[{'max_abs':float((a-b).abs().max()),'mismatches':int((a!=b).sum()),'relative_l2':float((a.float()-b.float()).norm()/a.float().norm())} for a,b in zip(reference,out)]}
