from collections import Counter
from vllm.config import CUDAGraphMode


class Stage3Worker:
    def stage3_set_mode(self, mode):
        if mode not in ('full_piecewise', 'decode_only', 'eager'):
            raise ValueError(mode)
        runner = self.model_runner
        if not hasattr(self, '_stage3_dispatch'):
            self._stage3_dispatch = runner.cudagraph_dispatcher.dispatch
            worker = self

            def dispatch(*args, **kwargs):
                allowed = kwargs.get('valid_modes')
                if worker._stage3_mode == 'eager':
                    target = {CUDAGraphMode.NONE}
                elif worker._stage3_mode == 'decode_only':
                    target = {CUDAGraphMode.NONE, CUDAGraphMode.FULL}
                else:
                    target = None
                if target is not None:
                    kwargs['valid_modes'] = target if allowed is None else target & allowed
                result = worker._stage3_dispatch(*args, **kwargs)
                worker._stage3_counts[str(result[0])] += 1
                worker._stage3_shapes[str(result)] += 1
                return result

            runner.cudagraph_dispatcher.dispatch = dispatch
        self._stage3_mode = mode
        self._stage3_counts = Counter()
        self._stage3_shapes = Counter()
        return {'mode': mode, 'runner': type(runner).__name__}

    def stage3_stats(self):
        runner = self.model_runner
        return {'mode': self._stage3_mode,
                'dispatch_counts': dict(self._stage3_counts),
                'dispatch_shapes': dict(self._stage3_shapes),
                'config_mode': str(runner.compilation_config.cudagraph_mode),
                'capture_sizes': runner.compilation_config.cudagraph_capture_sizes}
