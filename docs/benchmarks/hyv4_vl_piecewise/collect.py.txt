import hashlib
import json
from pathlib import Path

r = Path('/root/hy4-perf-stage3')
load = lambda path: json.loads((r/path).read_text())
read_rows = lambda path: [json.loads(x) for x in (r/path).read_text().splitlines()]
full = load('accuracy-full_piecewise/summary.json')
decode = load('accuracy-decode_only/summary.json')
assert all(s['complete'] and s['completed'] == 765 for s in (full, decode))
assert load('accuracy-full_piecewise/manifest.json') == load('accuracy-decode_only/manifest.json')
a = {x['index']: x for x in read_rows('accuracy-full_piecewise/results.jsonl')}
b = {x['index']: x for x in read_rows('accuracy-decode_only/results.jsonl')}
assert a.keys() == b.keys() == set(range(765))
rows, changes = [], []
for i in sorted(a):
    x, y = a[i], b[i]
    assert (x['image_sha256'], x['question'], x['answer']) == (y['image_sha256'], y['question'], y['answer'])
    row = {'index': i, 'image_sha256': x['image_sha256'], 'question': x['question'], 'reference':x['answer'],
           'full_piecewise_prediction':x['prediction'], 'decode_only_prediction':y['prediction'],
           'full_piecewise_correct':x['correct'], 'decode_only_correct':y['correct'],
           'full_piecewise_strict_correct':x['strict_correct'], 'decode_only_strict_correct':y['strict_correct'],
           'full_piecewise_seconds':x['seconds'], 'decode_only_seconds':y['seconds']}
    rows.append(row)
    if x['prediction'] != y['prediction']:
        changes.append(row)
errors = {mode:{k:s[k] for k in ['http_errors','empty_answers','truncated']} for mode,s in [('full_piecewise',full),('decode_only',decode)]}
accuracy = {'items':765,'full_piecewise_correct':full['correct'],'decode_only_correct':decode['correct'],
            'full_piecewise_strict_correct':full['strict_correct'],'decode_only_strict_correct':decode['strict_correct'],
            'raw_changed':len(changes),'gained':sum(x['full_piecewise_correct'] and not x['decode_only_correct'] for x in changes),
            'regressed':sum(x['decode_only_correct'] and not x['full_piecewise_correct'] for x in changes),
            'errors':errors,'changes':changes,
            'strict_acceptance_pass':not changes and all(v == 0 for d in errors.values() for v in d.values())}
(r/'accuracy-comparison.json').write_text(json.dumps(accuracy,ensure_ascii=False,indent=2)+'\n')
(r/'paired-predictions.jsonl').write_text(''.join(json.dumps(row,ensure_ascii=False)+'\n' for row in rows))
timings={}
for suite in ['qa_c1','qa_c2','caption32_c1','two_images32_c1']:
    after = load('e2e-full_piecewise-repeat/summary.json')[suite]
    entry={}
    for label,where in [('same_process_decode_only','e2e-decode_only-repeat'),('same_process_eager','e2e-eager')]:
        before=load(where+'/summary.json')[suite]
        entry[label]={'p50_before_ms':before['latency_p50_seconds']*1000,'p50_after_ms':after['latency_p50_seconds']*1000,
                      'ttft_before_ms':before['ttft_p50_seconds']*1000,'ttft_after_ms':after['ttft_p50_seconds']*1000,
                      'rps_before':before['requests_per_second'],'rps_after':after['requests_per_second'],
                      'throughput_increase_percent':(after['requests_per_second']/before['requests_per_second']-1)*100,
                      'p50_reduction_percent':(1-after['latency_p50_seconds']/before['latency_p50_seconds'])*100}
        x,y=load(where+'/'+suite+'.json'),load('e2e-full_piecewise-repeat/'+suite+'.json')
        assert [(z['index'],z['second_index']) for z in x]==[(z['index'],z['second_index']) for z in y]
        diffs=[{'index':u['index'],'second_index':u['second_index'],'before':u['answer'],'after':v['answer']} for u,v in zip(x,y) if u['answer']!=v['answer']]
        if '32' in suite:
            assert all(z['usage']['completion_tokens']==32 for z in x+y)
        entry[label]['raw_changed']=len(diffs)
        entry[label]['changes']=diffs
    entry['requests']=after['requests']
    timings[suite]=entry
(r/'performance-comparison.json').write_text(json.dumps(timings,ensure_ascii=False,indent=2)+'\n')
mode_checks={}
for mode in ['full_piecewise','decode_only','eager']:
    label='accuracy-'+mode if mode!='eager' else 'e2e-eager'
    data=load(label+'-dispatch.json')['results']
    assert len(data)==8 and all(x==data[0] for x in data)
    counts=data[0]['dispatch_counts']
    assert data[0]['config_mode']=='FULL_AND_PIECEWISE'
    if mode=='full_piecewise': assert counts.get('PIECEWISE',0)>0 and counts.get('FULL',0)>0
    if mode=='decode_only': assert counts.get('FULL',0)>0 and 'PIECEWISE' not in counts
    if mode=='eager': assert set(counts)=={'NONE'}
    mode_checks[mode]=data[0]
(r/'dispatch-validation.json').write_text(json.dumps({'ranks':8,'passed':True,'per_rank':mode_checks},indent=2)+'\n')
acceptance={'full_qa_raw_check':accuracy['strict_acceptance_pass'],
            'all_timing_outputs_identical_vs_decode_only':all(x['same_process_decode_only']['raw_changed']==0 for x in timings.values()),
            'production_defaults_changed':False}
(r/'acceptance.json').write_text(json.dumps(acceptance,indent=2)+'\n')
print(json.dumps({'accuracy':accuracy,'acceptance':acceptance,'timing':timings},ensure_ascii=False,indent=2))

# Score the short timing suites with the unchanged pinned task scorer too.
import importlib.util
import sys
sys.path.insert(0, '/tmp/vllm-plugin-das-pr')
from tools.hyv4_realworldqa_eval import score
spec=importlib.util.spec_from_file_location('mcq_extract', '/root/hy4-realworldqa/reference/mcq_extract.py')
extractor=importlib.util.module_from_spec(spec);spec.loader.exec_module(extractor)
scored={}
for suite in ['qa_c1','qa_c2']:
 before=load('e2e-decode_only-repeat/'+suite+'.json')
 after=load('e2e-full_piecewise-repeat/'+suite+'.json')
 changes=[];bcorrect=acorrect=0
 for x,y in zip(before,after):
  assert x['reference']==y['reference']
  bc=score(x['reference'],x['answer'],extractor.extract_mcq_answer)[1]
  ac=score(y['reference'],y['answer'],extractor.extract_mcq_answer)[1]
  bcorrect+=bc;acorrect+=ac
  if x['answer']!=y['answer']:changes.append({'index':x['index'],'reference':x['reference'],'decode_only':x['answer'],'full_piecewise':y['answer'],'decode_only_correct':bc,'full_piecewise_correct':ac})
 scored[suite]={'requests':len(before),'decode_only_correct':bcorrect,'full_piecewise_correct':acorrect, 'gained':sum(x['full_piecewise_correct'] and not x['decode_only_correct'] for x in changes),'regressed':sum(x['decode_only_correct'] and not x['full_piecewise_correct'] for x in changes),'changes':changes}
(r/'scored-qa-comparison.json').write_text(json.dumps(scored,ensure_ascii=False,indent=2)+'\n')

fullsplit={'items':765, 'unique_encoded_images':len({x['image_sha256'] for x in rows}), 'concurrency':1, 'order':['full_piecewise','decode_only'], 'timing_includes':'client image decoding/RGB/PNG encoding, non-streaming HTTP and task scoring', 'modes':{}}
for mode in ['full_piecewise','decode_only']:
 summary=load('accuracy-'+mode+'/summary.json')
 logs=[json.loads(line) for line in (r/('accuracy-'+mode+'.log')).read_text().splitlines() if line.startswith('{')]
 assert logs[-1]['completed']==765
 wall=logs[-1]['seconds']
 fullsplit['modes'][mode]={'wall_seconds':wall, 'requests_per_second':765/wall, 'median_request_seconds':summary['median_request_seconds']}
f,d=fullsplit['modes']['full_piecewise'],fullsplit['modes']['decode_only']
fullsplit['throughput_increase_percent']=(f['requests_per_second']/d['requests_per_second']-1)*100
fullsplit['median_latency_reduction_percent']=(1-f['median_request_seconds']/d['median_request_seconds'])*100
(r/'fullsplit-performance.json').write_text(json.dumps(fullsplit,indent=2)+'\n')

previous={x['index']:x for x in map(json.loads,Path('/root/hy4-perf-stage2/graph-accuracy-c1/results.jsonl').read_text().splitlines())}
previous_changes=[]
for i,x in b.items():
 y=previous[i]
 assert (x['image_sha256'],x['question'],x['answer'])==(y['image_sha256'],y['question'],y['answer'])
 if x['prediction']!=y['prediction']:previous_changes.append({'index':i,'previous':y['prediction'],'current_decode_only':x['prediction']})
(r/'historical-control-comparison.json').write_text(json.dumps({'items':765,'protocol':'across-process decode-only c1, same source kernels/data/prompts/scorer; distinct graph allocation/config controls','raw_changed':len(previous_changes),'previous_correct':sum(x['correct'] for x in previous.values()),'current_correct':decode['correct'],'changes':previous_changes},ensure_ascii=False,indent=2)+'\n')
