"""Offline TP0 gate/up GEMM tuning probe; run only after the serving process exits.

Uses actual layer-1 checkpoint weights, synthetic BF16 inputs and synthetic
unique-per-token top-8 routing. This is not a complete expert/model accuracy test.
"""
import argparse
import hashlib
import json
import statistics
import os
from pathlib import Path

import torch
import triton.language as tl
from safetensors import safe_open
from vllm import _custom_ops as ops
from vllm.model_executor.layers.fused_moe.fused_moe import (
    _prepare_expert_assignment,
    get_default_config,
    invoke_fused_moe_triton_kernel,
)

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--output-dir', type=Path, required=True)
args = parser.parse_args()
args.output_dir.mkdir(parents=True, exist_ok=True)
root = Path('/module/Hy4-testing-Channel-FP8-w8a8')
index = json.loads((root/'model.safetensors.index.json').read_text())['weight_map']
E, N, K, topk = 256, 512, 6144, 8
# TP0 owns the first 256 intermediate rows of each gate and up projection.
w = torch.empty((E, N, K), dtype=torch.float8_e4m3fn, device='cpu')
scales = torch.empty((E, N, 1), dtype=torch.float32, device='cpu')
weight_hash = hashlib.sha256()
for e in range(E):
    for kind, begin in [('gate', 0), ('up', 256)]:
        prefix = f'model.layers.1.mlp.experts.{e}.{kind}_proj.'
        for suffix, dest in [('weight', w), ('weight_scale', scales)]:
            name = prefix+suffix
            with safe_open(root/index[name], framework='pt', device='cpu') as f:
                value = f.get_slice(name)[:256]
            dest[e, begin:begin+256].copy_(value)
            weight_hash.update(value.contiguous().view(torch.uint8).numpy().tobytes())
B, Bscale = w.cuda(), scales.cuda()
del w, scales
results = []
for M in [1, 2, 128, 256, 512, 768, 1024]:
    torch.manual_seed(338+M)
    x = torch.randn((M, K), device='cuda', dtype=torch.bfloat16)
    A, Ascale = ops.scaled_fp8_quant(x, use_per_token_if_dynamic=True)
    ids = torch.stack([torch.randperm(E, device='cuda')[:topk] for _ in range(M)]).to(torch.int32)
    baseline = get_default_config(M, E, 256, K, topk, 'fp8_w8a8', None)
    configs = [baseline]
    for bm in [16, 32, 64]:
        for bn in [64, 128]:
            for warps in [4, 8]:
                cfg = dict(baseline, BLOCK_SIZE_M=bm, BLOCK_SIZE_N=bn, num_warps=warps)
                if cfg not in configs:
                    configs.append(cfg)
    ref = None
    samples = []
    for cfg in configs:
        C = torch.zeros((M, topk, N), dtype=torch.bfloat16, device='cuda')
        sorted_ids, expert_ids, num_padded = _prepare_expert_assignment(ids, cfg, M, topk, E, None)
        def forward():
            invoke_fused_moe_triton_kernel(
                A, B, C, Ascale, Bscale, None, sorted_ids, expert_ids,
                num_padded, False, topk, cfg, tl.bfloat16,
                use_fp8_w8a8=True, use_int8_w8a8=False,
                use_int8_w8a16=False, use_int4_w4a16=False,
                per_channel_quant=True)
        try:
            for _ in range(3):
                forward()
            torch.cuda.synchronize()
            if ref is None:
                assert torch.isfinite(C).all().item() and C.float().norm().item() > 0
                ref = C.clone()
            exact = torch.equal(C, ref)
            max_abs = (C.float()-ref.float()).abs().max().item()
            relative_l2 = ((C.float()-ref.float()).norm()/ref.float().norm()).item()
            graph = torch.cuda.CUDAGraph()
            with torch.cuda.graph(graph):
                for _ in range(8):
                    forward()
            graph.replay()
            torch.cuda.synchronize()
            measured = []
            for _ in range(5):
                start, end = torch.cuda.Event(enable_timing=True), torch.cuda.Event(enable_timing=True)
                start.record()
                for _ in range(8):
                    graph.replay()
                end.record()
                end.synchronize()
                measured.append(start.elapsed_time(end)/64)
            samples.append({'config':cfg, 'exact':exact, 'max_abs':max_abs,
                            'relative_l2':relative_l2, 'kernel_ms':statistics.median(measured), 'kernel_ms_rounds':measured})
            del graph
        except Exception as exc:
            samples.append({'config':cfg, 'error':str(exc)})
        print(json.dumps({'M':M, **samples[-1]}), flush=True)
    valid = [x for x in samples if x.get('exact')]
    best = min(valid, key=lambda x:x['kernel_ms'])
    before = samples[0]
    results.append({'M':M, 'baseline':before, 'fastest_exact':best,
                    'speedup':before['kernel_ms']/best['kernel_ms'], 'candidates':samples})
    (args.output_dir/'results.json').write_text(json.dumps(results,indent=2)+'\n')
manifest = {'probe':'TP0 layer-1 gate/up GEMM only; fixed-K tile 128; alternate M/N tiles and warp counts',
            'weights':'actual checkpoint first TP shard, all 256 experts',
            'weights_scales_sha256':weight_hash.hexdigest(),
            'inputs':'synthetic seeded BF16, native dynamic per-token FP8 quantization',
            'routing':'synthetic top8, unique experts per token',
            'timing':'CUDA-event median of five rounds, 64 kernel calls/round under graph replay',
            'excluded':'activation/clamp, down projection, routing cost, expert alignment cost, TP reduction and HTTP',
            'deployment_changed':False,
            'environment':{k:os.environ.get(k) for k in ['VLLM_BATCH_INVARIANT','VLLM_TUNED_CONFIG_FOLDER','VLLM_ROCM_USE_AITER']},
            'software':{'torch':torch.__version__}, 'device':torch.cuda.get_device_name(0)}
(args.output_dir/'manifest.json').write_text(json.dumps(manifest,indent=2)+'\n')
