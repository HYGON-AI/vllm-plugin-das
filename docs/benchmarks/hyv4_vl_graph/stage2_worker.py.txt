from collections import Counter
from vllm.config import CUDAGraphMode


class Stage2Worker:
    def stage2_set_mode(self, mode):
        if mode not in ('graph', 'eager'):
            raise ValueError(mode)
        runner = self.model_runner
        if not hasattr(self, '_stage2_dispatch'):
            self._stage2_dispatch = runner.cudagraph_dispatcher.dispatch
            worker = self
            def dispatch(*args, **kwargs):
                if worker._stage2_mode == 'eager':
                    kwargs['valid_modes'] = {CUDAGraphMode.NONE}
                result = worker._stage2_dispatch(*args, **kwargs)
                worker._stage2_counts[str(result[0])] += 1
                return result
            runner.cudagraph_dispatcher.dispatch = dispatch
        self._stage2_mode = mode
        self._stage2_counts = Counter()
        return {'mode': mode, 'runner': type(runner).__name__}

    def stage2_stats(self):
        return {'mode': self._stage2_mode, 'dispatch_counts': dict(self._stage2_counts)}
