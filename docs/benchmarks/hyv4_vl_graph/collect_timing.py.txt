import json
from pathlib import Path
r=Path('/root/hy4-perf-stage2')
def get(where):return json.loads((where/'summary.json').read_text())
graph=get(r/'e2e-graph-repeat');base=get(r/'e2e-eager-control');history=get(Path('/root/hy4-vit-perf/e2e-optimized'))
result={}
for name,after in graph.items():
 entry={}
 for label,before in [('same_process_eager',base[name]),('stage1_eager',history[name])]:
  entry[label]={'latency_before_ms':before['latency_p50_seconds']*1000,'latency_after_ms':after['latency_p50_seconds']*1000,'latency_reduction_percent':(1-after['latency_p50_seconds']/before['latency_p50_seconds'])*100,'throughput_before_rps':before['requests_per_second'],'throughput_after_rps':after['requests_per_second'],'throughput_increase_percent':(after['requests_per_second']/before['requests_per_second']-1)*100,'ttft_before_ms':before['ttft_p50_seconds']*1000,'ttft_after_ms':after['ttft_p50_seconds']*1000}
 a=json.loads((r/'e2e-eager-control'/f'{name}.json').read_text());b=json.loads((r/'e2e-graph-repeat'/f'{name}.json').read_text());assert [(x['index'],x['second_index']) for x in a]==[(x['index'],x['second_index']) for x in b]
 entry['raw_answers_changed']=sum(x['answer']!=y['answer'] for x,y in zip(a,b));entry['requests']=len(a)
 if '32' in name:assert all(x['usage']['completion_tokens']==32 for x in a+b)
 result[name]=entry
(r/'performance-comparison.json').write_text(json.dumps(result,indent=2));print(json.dumps(result,indent=2))
