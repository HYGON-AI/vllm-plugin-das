import json
import subprocess
import time
from pathlib import Path
import requests
r=Path('/root/hy4-perf-stage2')
while True:
 log=(r/'repeat-unprofiled-pipeline.log').read_text()
 if 'DONE graph repeat timing' in log:break
 if 'Traceback' in log or 'Traceback' in (r/'unprofiled-pair.log').read_text():raise RuntimeError('paired pipeline failed')
 time.sleep(10)
def rpc(method,*args):
 reply=requests.post('http://127.0.0.1:8010/collective_rpc',json={'method':method,'args':list(args),'timeout':120},timeout=150)
 reply.raise_for_status();return reply.json()
(r/'graph-final-smoke-mode.json').write_text(json.dumps(rpc('stage2_set_mode','graph'),indent=2))
for endpoint in ['reset_encoder_cache','reset_mm_cache']:
 reply=requests.post('http://127.0.0.1:8010/'+endpoint,timeout=60);reply.raise_for_status()
with (r/'smoke-graph-final.log').open('w') as out:
 subprocess.run(['python','tools/hyv4_vl_image_smoke.py','--output-dir',str(r/'smoke-graph-final'),'--photo','/root/hy4-vl-validation/cherry_blossom.jpg'],stdout=out,stderr=subprocess.STDOUT,check=True)
(r/'graph-final-smoke-dispatch.json').write_text(json.dumps(rpc('stage2_stats'),indent=2))
print('DONE final smoke',flush=True)
