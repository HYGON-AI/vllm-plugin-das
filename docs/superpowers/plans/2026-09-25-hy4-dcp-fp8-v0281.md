import os


def validate_hy4_dcp_config(vllm_config: object) -> bool:
    model = vllm_config.model_config
    parallel = vllm_config.parallel_config
    if model.architectures != ["HYV4ForCausalLM"]:
        return False
    if parallel.decode_context_parallel_size == 1:
        return False
    topology = (
        parallel.tensor_parallel_size,
        parallel.decode_context_parallel_size,
        parallel.prefill_context_parallel_size,
        parallel.pipeline_parallel_size,
        parallel.data_parallel_size,
    )
    if topology != (8, 2, 1, 1, 1):
        raise ValueError(f"HY V4 DCP topology is not validated: {topology}")
    if vllm_config.cache_config.cache_dtype not in (
        "fp8_e4m3", "fp8_ds_mla"
    ):
        raise ValueError("HY V4 DCP requires --kv-cache-dtype fp8_e4m3")
    if not vllm_config.use_v2_model_runner or not model.use_mla:
        raise ValueError("HY V4 DCP requires MLA Model Runner V2")
    if model.hf_config.num_attention_heads != 64:
        raise ValueError("HY V4 DCP requires 64 attention heads")
    local_heads = model.hf_config.num_attention_heads // parallel.tensor_parallel_size
    gathered_heads = local_heads * parallel.decode_context_parallel_size
    if (local_heads, gathered_heads) != (8, 16):
        raise ValueError("HY V4 DCP FP8 head envelope is not validated")
    if (parallel.dcp_comm_backend != "ag_rs" or
            parallel.cp_kv_cache_interleave_size != 1):
        raise ValueError("HY V4 DCP requires ag_rs and interleave size 1")
    if not parallel.enable_expert_parallel:
        raise ValueError("HY V4 DCP requires expert parallelism")
    if vllm_config.kernel_config.moe_backend != "aiter":
        raise ValueError("HY V4 DCP requires the selected AITER MoE backend")
    qrep_env = os.getenv("VLLM_DCP_Q_REPLICATE", "0").strip().lower()
    if parallel.dcp_q_replicate or qrep_env in {"1", "true", "yes", "on"}:
        raise ValueError("HY V4 DCP query replication is not validated")
    speculative = vllm_config.speculative_config
    if speculative is not None and (
        speculative.method != "mtp" or
        speculative.num_speculative_tokens != 3
    ):
        raise ValueError("HY V4 DCP supports only built-in MTP3")
    if (vllm_config.lora_config is not None or
            model.is_multimodal_model or
            vllm_config.cache_config.kv_offloading_size is not None or
            getattr(vllm_config.kv_transfer_config, "kv_connector", None)):
        raise ValueError("HY V4 DCP excludes LoRA, MM, offload and P/D")
    from vllm_hcu.patch.config import get_hcu_config
    features = get_hcu_config(vllm_config)
    if features.enable_multi_layers_mtp or features.enable_lightly_cp:
        raise ValueError("HY V4 DCP excludes multi-layer MTP and lightly-CP")
    return True

def test_hyv4_dcp_gathers_loaded_sinks_once(monkeypatch):
    from vllm_hcu.models.hy_v4 import hcu_sparse

    impl = _bare_sparse_impl(torch.arange(8, dtype=torch.float32))
    impl.dcp_world_size = 2
    other = torch.arange(8, 16, dtype=torch.float32)
    calls = []

    class Group:
        def all_gather(self, value, dim):
            calls.append((value.clone(), dim))
            return torch.cat((value, other), dim=0)

    monkeypatch.setattr(hcu_sparse, "get_dcp_group", lambda: Group())
    impl.sinks.add_(10)

```python
from vllm.distributed.parallel_state import get_dcp_group


def process_weights_after_loading(self, act_dtype: torch.dtype) -> None:
    super().process_weights_after_loading(act_dtype)
    self._dcp_gathered_sinks = None
    if self.dcp_world_size > 1 and self.sinks is not None:
        self._dcp_gathered_sinks = get_dcp_group().all_gather(
            self.sinks.detach().contiguous(), dim=0
        )


def _fp8_sink_for_gathered_query(self, q: torch.Tensor) -> torch.Tensor | None:
    if self.dcp_world_size <= 1:
        return self._sinks_for_query(q, head_dim=2,
                                     kernel_heads=self.fp8_decode_padded_heads)
    if self.dcp_rank != 0:
        return None
    sinks = self._dcp_gathered_sinks
    if sinks is None or sinks.numel() != q.shape[2]:
        raise RuntimeError("HY V4 DCP sink does not match gathered query heads")
    padded = sinks.new_full((self.fp8_decode_padded_heads,), float("-inf"))
    padded[:sinks.numel()] = sinks
    return padded


def _forward_fp8_kv_mixed_batch(self, q, cache, topk_indices, metadata):
    output, raw_lse = super()._forward_fp8_kv_mixed_batch(
        q, cache, topk_indices, metadata
    )
    if self.dcp_world_size > 1 and self.dcp_rank == 0 and self.sinks is not None:
        if raw_lse is None or self._dcp_gathered_sinks is None:
            raise RuntimeError("HY V4 DCP requires loaded sink and per-head LSE")
        return output, torch.logaddexp(
            raw_lse, self._dcp_gathered_sinks.view(1, -1)
        )
    return output, raw_lse
```

@pytest.mark.parametrize("a", [0.1, float("-inf")])
def test_dcp_single_sink_matches_global_softmax(a):
    a, b, sink = torch.tensor([a, 0.7, 1.2], dtype=torch.float64)
    v0, v1 = torch.tensor([2.0, 4.0], dtype=torch.float64)
    rank0_lse = torch.logaddexp(a, sink)
    rank0_output = torch.where(torch.isneginf(a), 0.0,
                               torch.exp(a - rank0_lse) * v0)
    merged = (torch.exp(rank0_lse) * rank0_output + torch.exp(b) * v1) / (
        torch.exp(rank0_lse) + torch.exp(b)
    )
    oracle = (torch.exp(a) * v0 + torch.exp(b) * v1) / (
        torch.exp(a) + torch.exp(b) + torch.exp(sink)
    )
    torch.testing.assert_close(merged, oracle)
    double_sink = (torch.exp(a) * v0 + torch.exp(b) * v1) / (
        torch.exp(a) + torch.exp(b) + 2 * torch.exp(sink)
    )
    assert not torch.isclose(double_sink, oracle)
